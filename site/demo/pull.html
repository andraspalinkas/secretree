+package main
+
+import "time"
+
+// retry calls fn up to n times, sleeping between attempts.
+func retry(n int, delay time.Duration, fn func() error) error {
+ var err error
+ for i := 0; i < n; i++ {
+ if err = fn(); err == nil {
+ return nil
+ }
+ time.Sleep(delay)
+ }
+ return err
+}
+
+// backoff doubles the delay after each failure.
+func backoff(d time.Duration) time.Duration { return d * 2 }
